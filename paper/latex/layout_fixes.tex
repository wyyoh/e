\newcommand{\figplaceholder}[2][5.2cm]{%
  \begin{figure}[H]\centering

% Keep long original displays inside the template's text block.
\long\def\[#1\]{%
  \begin{equation*}
    \begin{adjustbox}{max width=.98\linewidth}
      $\displaystyle #1$
    \end{adjustbox}
  \end{equation*}}
